```dot
         label="本番向け宣言構成 r17\n対象: 2026-09-30 / 本番稼働は未観測"];
  node [shape=box, fontname="Noto Sans CJK JP", fontsize=12];
  edge [fontname="Noto Sans CJK JP", fontsize=11];
  subgraph cluster_B1 {
    label="B1: 環境ラベル 本番向け（宣言）\nネットワーク・信頼境界を意味しない [S1]";
    N1 [label="N1 受付処理（宣言）\n2複製を指定 [S1,S2]"];
    N2 [label="N2 記録庫（宣言）\n保存先 [S1,S2]"];
    N1 -> N2 [label="E1: 記録を書き込む（宣言） [S1,S2]"];
  }

```
